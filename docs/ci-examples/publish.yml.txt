# Configure npm trusted publishing only after authenticated validation.
# A version tag is an explicit release action; this workflow is not run by PRs.
name: Publish
on:
  push:
    tags: ['v*.*.*']
permissions:
  contents: read
jobs:
  publish:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      id-token: write
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '22.22.2'
          registry-url: https://registry.npmjs.org
          cache: npm
      - run: npm ci --ignore-scripts
      - run: npm run lint
      - run: npm test
      - run: npm publish --ignore-scripts --provenance --access public
